async fn read_enabled_providers_async(app: &tauri::AppHandle) -> Vec<(String, String)> {
    use tauri::Manager;
    let app_state = match app.try_state::<Arc<AppState>>() {
        Some(s) => s.inner().clone(),
        None => return Vec::new(),
    };
    match tauri::async_runtime::spawn_blocking(move || {
        let _gate = app_state.data_gate.read();
        let db = match require_ready_gated(&app_state, &_gate) {
            Ok(d) => d,
            Err(_) => return Vec::new(),
        };
        db.with_conn(|conn| {
            let list = db_providers::list(conn)?;
            Ok::<_, DbErr>(list.into_iter().filter(|p| p.enabled).map(|p| (p.uuid, p.name)).collect::<Vec<_>>())
        })
    })
    .await
    {
        Ok(Ok(v)) => v,
        _ => Vec::new(),
    }
}

async fn read_primary_status_async(app: &tauri::AppHandle) -> String {
    use tauri::Manager;
    let app_state = match app.try_state::<Arc<AppState>>() {
        Some(s) => s.inner().clone(),
        None => return "No provider".into(),
    };
    match tauri::async_runtime::spawn_blocking(move || {
        let _gate = app_state.data_gate.read();
        let db = match require_ready_gated(&app_state, &_gate) {
            Ok(d) => d,
            Err(_) => return "No provider".to_string(),
        };
        let selection = db.with_conn(|conn| db_providers::read_active_selection(conn));
        match selection {
            Ok(sel) => match sel.primary {
                Some(uuid) => {
                    let name = db.with_conn(|conn| db_providers::get(conn, &uuid)).ok().map(|p| p.name);
                    name.unwrap_or_else(|| "Unknown provider".into())
                }
                None => "No provider".into(),
            },
            Err(_) => "No provider".into(),
        }
    })
    .await
    {
        Ok(s) => s,
        Err(_) => "No provider".into(),
    }
}

### Task A3: P1-3 — Provider switch FIFO worker (last-click-wins at DB level)

**Files:**
- Modify: `src-tauri/src/lib.rs` lines 1540-1605 (db_set_active_primary + handle_switch_provider_core + handle_switch_provider) + lines 2744-2755 (handle_tray_menu_event switch arm)
- Test: `src-tauri/tests/switch_concurrency.rs` **(new)**

**Interfaces:**
- Produces: a `SwitchWorker` (or revision-guarded DB path) that ensures the last switch click's DB commit is the final state.

**Decision: revision-guarded DB path (Option B from the audit).** Rationale: a FIFO worker adds a long-lived background task + channel that is harder to test and reason about than a simple revision check. The revision already exists on the tray controller; we extend it to guard the DB write. `handle_switch_provider_core` captures `rev` via `begin_switch()`, then `db_set_active_primary` receives `rev` and checks it against `app_state.tray.lock().switch_revision()` BEFORE the DB commit. If `rev != switch_revision`, the DB write is skipped (a newer click already bumped the revision).

    let db_path = dir.path().join("linguaray.db");
    let db = Database::open(&db_path).expect("Database::open");
    db.with_conn(|conn| {
        let tx = conn.transaction()?;
        schema::create_all_tables(&tx)?;
        schema::seed_singletons(&tx)?;
        tx.commit()?;
        Ok(())

    let renderer = Arc::new(RecordingRenderer::default());

        tray: Arc::new(parking_lot::Mutex::new(
